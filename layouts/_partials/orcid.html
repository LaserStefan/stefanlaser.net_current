{{- /* Works from the ORCID public API as slice of dict (title, year, venue, type, url, doi, source). Call with partialCached. */ -}} {{- $out := slice -}} {{- $types := dict "journal-article" "Journal article" "book" "Book" "book-chapter" "Book chapter" "edited-book" "Edited collection" "report" "Report" "working-paper" "Working paper" "conference-paper" "Conference paper" "book-review" "Review" "review" "Review" "magazine-article" "Magazine article" "newspaper-article" "Newspaper article" "dissertation-thesis" "Dissertation" "online-resource" "Online" "lecture-speech" "Talk" "other" "Other" -}} {{- with hugo.Data.site.publications.orcid -}} {{- $api := site.Params.orcidapi | default "https://pub.orcid.org/v3.0" -}} {{- $url := printf "%s/%s/works" $api (trim . " ") -}} {{- with try (resources.GetRemote $url (dict "headers" (dict "Accept" "application/json"))) -}} {{- with .Err }}{{ warnf "ORCID: %s" . }} {{- else with .Value -}} {{- range (.Content | transform.Unmarshal).group -}} {{- $w := index (index . "work-summary") 0 -}} {{- $title := "" }}{{ with $w.title }}{{ with .title }}{{ $title = .value }}{{ end }}{{ end -}} {{- if not $title }}{{ continue }}{{ end -}} {{- $year := "" }}{{ with index $w "publication-date" }}{{ with .year }}{{ $year = .value }}{{ end }}{{ end -}} {{- $venue := "" }}{{ with index $w "journal-title" }}{{ $venue = .value }}{{ end -}} {{- $doi := "" }}{{ $link := "" -}} {{- with index $w "external-ids" }}{{ range index . "external-id" -}} {{- if and (eq (index . "external-id-type") "doi") (not $doi) }}{{ $doi = index . "external-id-value" }}{{ end -}} {{- if not $link }}{{ with index . "external-id-url" }}{{ $link = .value }}{{ end }}{{ end -}} {{- end }}{{ end -}} {{- with $w.url }}{{ with .value }}{{ $link = . }}{{ end }}{{ end -}} {{- $doi = replaceRE `^(https?://(dx\.)?doi\.org/|doi:)` "" $doi -}} {{- if $doi }}{{ $link = printf "https://doi.org/%s" $doi }}{{ end -}} {{- $type := index $types ($w.type | default "other") | default (replace ($w.type | default "") "-" " " | strings.FirstUpper) -}} {{- $out = $out | append (dict "title" $title "year" $year "venue" $venue "type" $type "url" $link "doi" $doi "source" "orcid") -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
