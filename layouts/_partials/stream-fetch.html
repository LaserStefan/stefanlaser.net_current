{{- /* Posts from one service, fetched at build time. Param: "bluesky" or "mastodon". Returns dict: items, profile (link), label (@handle), configured (bool). Call with partialCached. */ -}} {{- $source := . -}} {{- $st := hugo.Data.site.stream -}} {{- $items := slice -}} {{- $profile := "" -}} {{- if eq $source "bluesky" -}} {{- with $st.bluesky_handle -}} {{- $handle := strings.TrimPrefix "@" . -}} {{- $profile = printf "https://bsky.app/profile/%s" $handle -}} {{- $filter := cond $st.hide_replies "posts_no_replies" "posts_with_replies" -}} {{- $url := printf "https://public.api.bsky.app/xrpc/app.bsky.feed.getAuthorFeed?actor=%s&filter=%s&limit=50" $handle $filter -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err }}{{ warnf "Bluesky stream: %s" . }} {{- else with .Value }} {{- range (.Content | transform.Unmarshal).feed -}} {{- if and $st.hide_reposts .reason }}{{ continue }}{{ end -}} {{- $p := .post -}} {{- $rkey := index (last 1 (split $p.uri "/")) 0 -}} {{- $item := dict "text" $p.record.text "date" (time.AsTime $p.record.createdAt) "url" (printf "https://bsky.app/profile/%s/post/%s" $p.author.handle $rkey) "repost" (cond (ne .reason nil) $p.author.displayName "") -}} {{- with $p.embed -}} {{- $t := index . "$type" -}} {{- if eq $t "app.bsky.embed.external#view" -}} {{- $item = merge $item (dict "quote_title" .external.title "quote_text" .external.description "quote_url" .external.uri) -}} {{- else if eq $t "app.bsky.embed.record#view" -}} {{- with .record -}} {{- if .author -}} {{- $item = merge $item (dict "quote_title" (.author.displayName | default .author.handle) "quote_text" (.value.text | default "")) -}} {{- end -}} {{- end -}} {{- else if eq $t "app.bsky.embed.images#view" -}} {{- $item = merge $item (dict "media" (len .images)) -}} {{- else if eq $t "app.bsky.embed.recordWithMedia#view" -}} {{- with .record.record }}{{ if .author }}{{ $item = merge $item (dict "quote_title" (.author.displayName | default .author.handle) "quote_text" (.value.text | default "")) }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- $items = $items | append $item -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- if and $st.mastodon_instance $st.mastodon_user -}} {{- $inst := strings.TrimPrefix "https://" $st.mastodon_instance | strings.TrimSuffix "/" -}} {{- $user := strings.TrimPrefix "@" $st.mastodon_user -}} {{- $profile = printf "https://%s/@%s" $inst $user -}} {{- with try (resources.GetRemote (printf "https://%s/api/v1/accounts/lookup?acct=%s" $inst $user)) -}} {{- with .Err }}{{ warnf "Mastodon lookup: %s" . }} {{- else with .Value -}} {{- $id := (.Content | transform.Unmarshal).id -}} {{- $url := printf "https://%s/api/v1/accounts/%s/statuses?limit=40&exclude_replies=%t&exclude_reblogs=%t" $inst $id (not (not $st.hide_replies)) (not (not $st.hide_reposts)) -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err }}{{ warnf "Mastodon stream: %s" . }} {{- else with .Value -}} {{- range (.Content | transform.Unmarshal) -}} {{- $s := . -}} {{- $repost := "" -}} {{- with .reblog }}{{ $s = . }}{{ $repost = .account.display_name | default .account.acct }}{{ end -}} {{- $text := $s.content | replaceRE `

\s*

` "\n\n" | replaceRE `` "\n" | plainify | htmlUnescape -}} {{- with $s.spoiler_text }}{{ $text = printf "CW: %s" . }}{{ end -}} {{- $item := dict "text" $text "date" (time.AsTime .created_at) "url" .url "repost" $repost -}} {{- with $s.card }}{{ $item = merge $item (dict "quote_title" .title "quote_text" (.provider_name | default "") "quote_url" .url) }}{{ end -}} {{- with $s.media_attachments }}{{ $item = merge $item (dict "media" (len .)) }}{{ end -}} {{- $items = $items | append $item -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end }} {{- $label := "" -}} {{- if eq $source "bluesky" }}{{ with $st.bluesky_handle }}{{ $label = printf "@%s" (strings.TrimPrefix "@" .) }}{{ end -}} {{- else if and $st.mastodon_user $st.mastodon_instance }}{{ $label = printf "@%s@%s" (strings.TrimPrefix "@" $st.mastodon_user) (strings.TrimPrefix "https://" $st.mastodon_instance | strings.TrimSuffix "/") }}{{ end -}} {{- return dict "items" $items "profile" $profile "label" $label "configured" (ne $profile "") -}}
