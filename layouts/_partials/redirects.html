Call once with partialCached. */ -}} {{- $map := slice -}} {{- range where site.RegularPages "Section" "projects" -}} {{- $p := . -}} {{- with $p.Params.old_url }}{{ $map = $map | append (dict "from" . "to" $p.RelPermalink) }}{{ end -}} {{- end -}} {{- $map = $map | append (dict "from" "/archive/index.html" "to" ("notes/" | relURL)) -}} {{- range $map -}} {{- $to := .to -}} {{- $from := strings.TrimPrefix "/" .from -}} {{- if hasSuffix $from "/" }}{{ $from = printf "%sindex.html" $from }}{{ end -}} {{- $html := printf `MovedThis page has moved.` (absURL $to) $to $to -}} {{- $_ := (resources.FromString $from $html).RelPermalink -}} {{- end -}}
